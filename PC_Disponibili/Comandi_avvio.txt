  MSI

  # Avvio (singoli apici — funziona da PowerShell senza errori &&)
  ssh giand@msi 'wmic process call create "cmd.exe /c cd /D ""C:\Users\giand\Desktop\prog\FCGS"" && ""venv\Scripts\python.exe"" federated_grid_search.py"'

  # PID (aspetta 5 secondi poi leggi)
  ssh giand@msi "powershell Get-Content 'C:\Users\giand\Desktop\prog\FCGS\log\log_MSI\bootstrap.log'"

  # Log in tempo reale
  ssh giand@msi "powershell Get-Content 'C:\Users\giand\Desktop\prog\FCGS\log\log_MSI\main.log' -Wait -Tail 30"

  # Kill FCGS (solo federated_grid_search, non altri Python)
  ssh giand@msi 'wmic process where "Name=''python.exe'' and CommandLine like ''%federated_grid_search%''" delete'
