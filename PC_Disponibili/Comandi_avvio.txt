============================================================
COMANDI DA ESEGUIRE DIRETTAMENTE SUL PC (dopo ssh <pc>)
============================================================
Una volta dentro la sessione SSH, usa questi comandi — nessun problema di quoting.

 ---
  MSI  (path: C:\Users\giand\Desktop\prog\FCGS)

  # Avvio
  cd C:\Users\giand\Desktop\prog\FCGS
  powershell Start-Process -FilePath venv\Scripts\python.exe -ArgumentList federated_grid_search.py -WorkingDirectory C:\Users\giand\Desktop\prog\FCGS -WindowStyle Hidden

  # PID (aspetta 5-10 secondi)
  powershell Get-Content log\log_MSI\bootstrap.log

  # Log in tempo reale
  powershell Get-Content log\log_MSI\main.log -Wait -Tail 30

  # Kill solo FCGS (non altri Python)
  powershell "Get-CimInstance Win32_Process | Where-Object {$_.Name -eq 'python.exe' -and $_.CommandLine -like '*federated_grid_search*'} | ForEach-Object {Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue}"
